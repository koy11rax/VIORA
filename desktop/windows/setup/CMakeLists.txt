cmake_minimum_required(VERSION 3.20)
project(VioraSetup LANGUAGES CXX RC)
set(VIORA_VERSION "1.0.0" CACHE STRING "Installer version")
if(NOT VIORA_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Invalid VIORA_VERSION")
endif()
string(REPLACE "." "," VIORA_VERSION_COMMA "${VIORA_VERSION}")
if(NOT EXISTS "${VIORA_MSI}" OR NOT EXISTS "${VIORA_ICON}")
  message(FATAL_ERROR "Provide the MSI and application icon")
endif()
configure_file(resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" @ONLY)
add_executable(Viora-Setup WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" setup.manifest)
target_compile_features(Viora-Setup PRIVATE cxx_std_17)
target_compile_options(Viora-Setup PRIVATE /utf-8)
target_compile_definitions(Viora-Setup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(Viora-Setup PRIVATE VIORA_SETUP_VERSION=L\"${VIORA_VERSION}\")
set_property(TARGET Viora-Setup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(Viora-Setup PRIVATE comctl32 shell32 ole32 advapi32)
